$$
\left\{
\begin{aligned}
x+y = 10 \\
2x+4y = 28
\end{aligned}
\right.
$$

$$
\begin{bmatrix}
1 & 1 \\
2 & 4 
\end{bmatrix}

\begin{bmatrix}
x \\
y 
\end{bmatrix}
=
\begin{bmatrix}
x & y \\
2x & 4y
\end{bmatrix}
=
\begin{bmatrix}
10 \\
28
\end{bmatrix}
$$
